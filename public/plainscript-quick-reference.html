06Absence and failure

define find(found: boolean) returns number?
    when found
        return 42
    end
    return nothing
end

try
    fail(7)
catch error: AppError
    print("{error.message} {error.code}")
end

Narrow an optional with when score is number before using it.

07Added in 2.3

class Counter
    value: number
    constructor(start: number)
        this.value = start
    end
    function next() -> number
        this.value += 1
        return this.value
    end
end

generator function pages(first: number, last: number) -> number
    let page = first
    while page <= last
        yield page
        page += 1
    end
end

08The loop: check, test, build, ship

  • plainscript check — types only, nothing runs
  • plainscript test — the project's own tests
  • plainscript build — emit modules and declarations
  • plainscript bundle — one ESM deployment artifact

The output is ordinary JavaScript modules, so npm packages and browser APIs still work.
